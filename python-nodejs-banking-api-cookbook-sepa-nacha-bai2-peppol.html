2. Python 3 — Generate 94-Character US NACHA Direct Deposit (`.ach`)

Download .py

Verifies 9-digit Federal Reserve ABA routing checksums (`3-7-1` MOD-10), computes entry hashes, and pads final output to strict `9999...` block-of-10 lines for Chase, BofA, Wells Fargo, Citi, and Silicon Valley Bank:

import json, os, urllib.request

url = f"https://api.apify.com/v2/acts/finops_studio~nacha-ach-file-generator/run-sync-get-dataset-items?token={os.environ['APIFY_TOKEN']}"
payload = {
    "immediateDestination": "021000021",
    "immediateDestinationName": "JPMORGAN CHASE",
    "immediateOrigin": "1999999999",
    "immediateOriginName": "ACME CORP INC",
    "companyName": "ACME CORP",
    "companyIdentification": "1999999999",
    "standardEntryClassCode": "PPD",
    "companyEntryDescription": "PAYROLL",
    "effectiveEntryDate": "261116",
    "originatingDfiIdentification": "02100002",
    "entries": [{
        "transactionCode": "22",
        "receivingDfiRouting": "026009593",
        "receivingAccountNumber": "9876543210",
        "amount": 3250.00,
        "individualId": "EMP-1042",
        "individualName": "JANE DOE"
    }]
}
req = urllib.request.Request(url, json.dumps(payload).encode(), {"Content-Type": "application/json"})
items = json.loads(urllib.request.urlopen(req).read())
open("payroll.ach", "w").write(items[0]["achContent"])
